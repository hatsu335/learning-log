## 2. Record & Comment 

```js
'use strict';  // 必ず先頭に表記する（エラーを出してくれる）

// お約束
console.log('hello world');


// const 定数 (値を変更できない)
const price = 150;
const rate = 1.1;

console.log(price * 120 * rate);
console.log(price * 130 * rate);
console.log(price * 140 * rate);


// let 変数
let l_price = 151;
let l_rate = 1.05;

l_price = 160;
l_rate = 1.08;

l_price++; // +1
l_price--; // -1

console.log(l_price * 120 * l_rate);
console.log(l_price * 130 * l_rate);
console.log(l_price * 140 * l_rate);


// 文字列
const f_name = 'Otani';
const l_name = 'Shohei';

console.log('I am ' + f_name + ' ' + l_name + '.');
// バッククォートで囲み、${定数}を使用
console.log(`I am ${f_name} ${l_name}.`);


// 文字列と数値
const n = prompt('Any number?');  // prompt = input
console.log(n + 10);           // prompt で得た値は文字列になる。結果も文字列になる
console.log(Number(n) + 10);   // Number() = int ※大文字に注意
                               // Number() に数値に変換できない値が入ると NaN を返す


// if 条件分岐 ※Pythonと同じ( > < >= <= ) Pythonと違う( === 等しい, !== 等しくない)
// 注意 1. {} の後に ; を付けない  2. else if = if else
const score = Number(prompt('Your Score?'));
if (score >= 90) {
    console.log('Rank A');
} else if (score >= 70) {
    console.log('Rank B');
} else {
    console.log('Rank C');
}


// switch 条件分岐
const color = prompt('Color?');
switch (color) {
    case 'red':
        console.log('Stop!');
        break;               // 全ての項目に break が必要
    case 'yellow':
        console.log('Slow...');
        break;
    case 'blue':
    case 'green':
        console.log('Go');
        break;
    default:                 // その他を表す
        console.log('???');
        break;
}
```
